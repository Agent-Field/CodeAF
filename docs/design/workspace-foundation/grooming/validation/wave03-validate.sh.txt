#!/usr/bin/env bash
# Focused functional validation for the local-work wave. Runs on Spark only.
set -uo pipefail
cd /home/santosh/src/aforge-personal-e2e
export GOMAXPROCS=4 GOFLAGS=-p=2
rev="$(git rev-parse HEAD)"
echo "REVISION $rev"
echo "TRACKED-CLEAN $(git status --porcelain --untracked-files=no | wc -l) modified tracked files"
echo "HOST $(hostname) $(uname -m) $(go version)"
echo "STARTED $(date -u +%FT%TZ)"
status=0
step() {
  local name="$1"; shift
  local t0=$(date +%s)
  echo "=== STEP $name: $*"
  if "$@"; then echo "=== RESULT $name PASS $(( $(date +%s) - t0 ))s"; else echo "=== RESULT $name FAIL $(( $(date +%s) - t0 ))s"; status=1; fi
}
step vet go vet ./internal/standing/ ./internal/session/ ./cmd/aforge/ ./internal/workspace/ ./internal/workspaceview/ ./internal/manual/
step vet-e2e go vet -tags e2e ./internal/e2e/ ./internal/session/
step small-suites go test -count=1 ./internal/standing/ ./internal/workspace/ ./internal/workspaceview/ ./internal/manual/
step session-selected go test -count=1 -timeout 15m -run 'Standing|Stand|ContextTrace|Governing|Organization|Trace|Firing|Inbox|Manual|Answer|Report|Rule|Finding|StopWhile|Publish|Correction|Withheld|Truncat|Digest|Hand|Journal|Torn|Session|Question|Mailbox|Fork|Deliver|Settle' ./internal/session/
step cmd-selected go test -count=1 -timeout 15m -run 'Help|Usage|Collections|Width|Tick|Command|Unknown|Doors|Vocabulary|Plain|Standing|Emptiness|Launch|Concept|Flag' ./cmd/aforge/
step e2e-untagged go test -count=1 ./internal/e2e/
step laws make test-laws
step changelog make changelog-check
step build make build
step packed-manual make test-packed-manual
step localwork-journey env AFORGE_LOCALWORK_KEEP=/tmp/opus-localwork/final-keep-$(git rev-parse --short HEAD) go test -tags e2e -count=1 -timeout 10m -run '^TestLocalWorkJourney$' -v ./internal/e2e/
echo "FINISHED $(date -u +%FT%TZ) STATUS $status"
exit $status
