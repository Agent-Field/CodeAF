#!/usr/bin/env bash
# Focused functional validation for wave 4. Runs on Spark only.
set -uo pipefail
cd /home/santosh/src/aforge-personal-e2e
export GOMAXPROCS=4 GOFLAGS=-p=2
rev="$(git rev-parse HEAD)"
echo "REVISION $rev"
echo "TRACKED-CLEAN $(git status --porcelain --untracked-files=no | wc -l) modified tracked files"
echo "HOST $(hostname) $(uname -m) $(go version)"
echo "STARTED $(date -u +%FT%TZ)"
status=0
step() {
  local name="$1"; shift
  local t0=$(date +%s)
  echo "=== STEP $name: $*"
  if "$@"; then echo "=== RESULT $name PASS $(( $(date +%s) - t0 ))s"; else echo "=== RESULT $name FAIL $(( $(date +%s) - t0 ))s"; status=1; fi
}
step vet go vet ./internal/standing/ ./internal/session/ ./cmd/aforge/ ./internal/workspace/ ./internal/workspaceview/ ./internal/manual/ ./internal/approval/
step vet-e2e go vet -tags e2e ./internal/e2e/ ./internal/session/
step small-suites go test -count=1 ./internal/standing/ ./internal/workspace/ ./internal/workspaceview/ ./internal/manual/ ./internal/approval/
step session go test -count=1 -timeout 15m ./internal/session/
step cmd go test -count=1 -timeout 15m ./cmd/aforge/
step e2e-untagged go test -count=1 ./internal/e2e/
step laws make test-laws
step changelog make changelog-check
step build make build
step packed-manual make test-packed-manual
step localwork-journey env AFORGE_LOCALWORK_KEEP=/tmp/opus-w4/final-keep-$(git rev-parse --short HEAD) go test -tags e2e -count=1 -timeout 10m -run '^TestLocalWorkJourney$' -v ./internal/e2e/
echo "FINISHED $(date -u +%FT%TZ) STATUS $status"
exit $status
